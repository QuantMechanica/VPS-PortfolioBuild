### Watch query

```sql
WITH chain(ea_id,symbol,q07_id,q08_rerun_target,q09_held) AS (
  VALUES
    ('QM5_12847','NDX.DWX','d5484501-e425-4c1d-941f-afa18706937d','2f4a4f40-80c2-4973-9909-d20dcad412d3','cc670aa2-c9b4-4605-aea3-a925afb238bf'),
    ('QM5_10706','GBPUSD.DWX','81cd341c-a2c1-4c18-9ec6-6b85f0226080','335d9197-0fef-4dbb-809f-36f616c84e78','cdfc4ddc-2f82-4321-ac36-876202eadcad'),
    ('QM5_12623','XAUUSD.DWX','1505ff12-34b9-4142-865e-869537fbd597','adff323e-5ff4-4af2-852a-7ffb12bdfd64','5302ac48-3123-4327-8d8a-506fffeee365')
)
SELECT c.ea_id,c.symbol,
       q7.status AS q07_status,q7.verdict AS q07_verdict,q7.evidence_path AS q07_evidence,
       q8.id AS regenerated_q08_id,q8.status AS q08_status,q8.verdict AS q08_verdict,
       c.q08_rerun_target,c.q09_held,
       q09new.id AS replacement_q09_id,q09new.status AS replacement_q09_status,
       q09new.verdict AS replacement_q09_verdict,h.active AS replacement_hold_active
FROM chain c
JOIN work_items q7 ON q7.id=c.q07_id
LEFT JOIN work_items q8
  ON q8.phase='Q08' AND q8.ea_id=c.ea_id AND q8.symbol=c.symbol
 AND json_extract(q8.payload_json,'$.append_only_rerun_of_work_item')=c.q08_rerun_target
 AND json_extract(q8.payload_json,'$.promoted_from_work_item')=c.q07_id
LEFT JOIN work_item_dependencies d
  ON d.parent_work_item_id=q8.id AND d.dependency_role='Q08_INPUT'
LEFT JOIN work_items q09new ON q09new.id=d.child_work_item_id AND q09new.phase='Q09_NEWS'
LEFT JOIN work_item_holds h ON h.work_item_id=q09new.id
ORDER BY c.ea_id;
```
