```text
r[i] = ln(C[i+1] / C[i]),                       i = 0..11

k = 0
for i = 0..11:
  for j = i..11:
    w[k] = (r[i] + r[j]) / 2
    k += 1

require k == 78
sorted = ascending copy of w[0..77]
hl = (sorted[38] + sorted[39]) / 2
```
